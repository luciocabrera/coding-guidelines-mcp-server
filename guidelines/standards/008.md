## 💻 Code Standard: Type-Safe Function Returns

### 🎯 Objective

To ensure all function return values are type-safe by preventing the return of `any` or `unknown` typed values from functions with explicit return types. This standard eliminates a major source of type safety holes, ensures predictable function contracts, and maintains type integrity throughout the codebase. Proper return type safety enables confident refactoring and prevents runtime type errors.

### 📜 Rule Details

1.  **Explicit Return Types:** All functions must have explicit return type annotations (unless return type is clearly inferred).
2.  **No Unsafe Returns:** Never return `any` or `unknown` typed values from functions with specific return types without proper type validation or narrowing.
3.  **Type Validation:** When returning external data (API responses, user input), validate and type the data before returning.
4.  **Consistent Types:** Return types must match the declared return type exactly - no implicit type widening or narrowing.
5.  **Generic Safety:** When using generics, ensure returned values match the generic constraints.

### 🔍 What is an Unsafe Return?

An unsafe return occurs when:

- A function returns a value typed as `any` but declares a specific return type
- A function returns a value typed as `unknown` without type validation
- A function's return type doesn't match what it actually returns
- Generic functions return values that don't satisfy their constraints
- Functions implicitly widen types unsafely

#### 4\. Type-Safe Error Handling with Returns

Ensure error handling maintains type safety in return values.

```typescript
// ✅ DO: Type-safe try-catch with proper return types
// File: safe-parse.util.ts

type SafeParseResult<T> =
  | {
      success: true;
      data: T;
    }
  | {
      success: false;
      error: Error;
    };

type SafeParseArgs<T> = {
  jsonString: string;
  validator: (data: unknown) => data is T;
};

export const safeParse = <T>({ jsonString, validator }: SafeParseArgs<T>): SafeParseResult<T> => {
  try {
    const parsed: unknown = JSON.parse(jsonString);

    if (!validator(parsed)) {
      return {
        success: false,
        error: new Error('Validation failed'),
      };
    }

    return {
      success: true,
      data: parsed, // Type-safe: validated as T
    };
  } catch (error) {
    return {
      success: false,
      error: error instanceof Error ? error : new Error('Unknown error'),
    };
  }
};

// Usage
type User = { id: string; name: string };

const isUser = (data: unknown): data is User => {
  if (typeof data !== 'object' || data === null) return false;
  const obj = data as Record<string, unknown>;
  return typeof obj.id === 'string' && typeof obj.name === 'string';
};

const result = safeParse({
  jsonString: '{"id":"123","name":"John"}',
  validator: isUser,
});

if (result.success) {
  console.log(result.data.name); // Type-safe: data is User
} else {
  console.error(result.error.message); // Type-safe: error is Error
}
```

```typescript
// ✅ DO: Type-safe async error handling
// File: safe-fetch.service.ts

type SafeFetchResult<T> =
  | {
      success: true;
      data: T;
    }
  | {
      success: false;
      error: Error;
    };

type SafeFetchArgs<T> = {
  url: string;
  validator: (data: unknown) => data is T;
};

export const safeFetch = async <T>({
  url,
  validator,
}: SafeFetchArgs<T>): Promise<SafeFetchResult<T>> => {
  try {
    const response = await fetch(url);

    if (!response.ok) {
      return {
        success: false,
        error: new Error(`HTTP error! status: ${response.status}`),
      };
    }

    const data: unknown = await response.json();

    if (!validator(data)) {
      return {
        success: false,
        error: new Error('Invalid response data'),
      };
    }

    return {
      success: true,
      data, // Type-safe: validated as T
    };
  } catch (error) {
    return {
      success: false,
      error: error instanceof Error ? error : new Error('Unknown error'),
    };
  }
};
```

```typescript
// ✅ DO: Type-safe function factory

type ValidationRule<T> = {
  validate: (value: T) => boolean;
  errorMessage: string;
};

type CreateValidatorArgs<T> = {
  rules: ValidationRule<T>[];
};

export const createValidator = <T>({ rules }: CreateValidatorArgs<T>) => {
  return (value: T): { isValid: boolean; errors: string[] } => {
    const errors: string[] = [];

    for (const rule of rules) {
      if (!rule.validate(value)) {
        errors.push(rule.errorMessage);
      }
    }

    return {
      isValid: errors.length === 0,
      errors,
    };
  }; // Type-safe: returns properly typed function
};

// Usage

  rules: [
    {
      validate: (email) => email.includes('@'),
      errorMessage: 'Email must contain @',
    },
    {
      validate: (email) => email.length > 5,
      errorMessage: 'Email must be at least 6 characters',
    },
  ],
});
```

```typescript
type RawUser = {
  id: string;
  name: string;
};

const isRawUser = (data: unknown): data is RawUser => {
  if (typeof data !== 'object' || data === null) return false;
  const obj = data as Record<string, unknown>;
  return typeof obj.id === 'string' && typeof obj.name === 'string';
};

export const mapUsers = (rawData: unknown[]): User[] => {
  const validatedData = rawData.filter(isRawUser);

  return validatedData.map(
    (item): User => ({
      id: item.id, // Type-safe: validated
      name: item.name, // Type-safe: validated
    }),
  );
};
```

#### 7\. Unsafe Optional Returns

Don't return potentially undefined values without proper typing.

```typescript
// ❌ DON'T: Return potentially undefined without proper type
// File: get-config-value.util.ts

type Config = {
  apiUrl?: string;
  timeout?: number;
};

export const getConfigValue = (config: Config, key: keyof Config): string | number => {
  return config[key]; // UNSAFE: might be undefined
};
```
